% ── خاص بالـ TP: مراحل سير الحصة التطبيقية ──────────────────────────────────
\newcommand{\tpstage}[3]{%
  \noindent
  \begin{tabularx}{\textwidth}{@{}>{\centering\arraybackslash}p{3.3cm}%
      >{\centering\arraybackslash}p{2.2cm}>{\raggedleft\arraybackslash}X@{}}
    \hline
    \cellcolor{primary}\color{white}\textbf{#1} &
    \cellcolor{primary}\color{white}\textbf{#2} &
    \begin{minipage}[t]{\linewidth}\raggedleft #3\end{minipage} \\
    \hline
  \end{tabularx}\par\vspace{-0.4pt}%
}

% ── خاص بالـ TP: جدول المهام التطبيقية ──────────────────────────────────────
\newcommand{\taskheader}{%
  \noindent
  \begin{tabularx}{\textwidth}{@{}>{\centering\arraybackslash}p{2.9cm}%
      >{\raggedleft\arraybackslash}X>{\raggedleft\arraybackslash}X@{}}
    \hline
    \cellcolor{primary}\color{white}\textbf{المهمة التطبيقية} &
    \cellcolor{primary}\color{white}\textbf{تعليمات الأستاذ والخطوات العملية} &
    \cellcolor{primary}\color{white}\textbf{عمل المتعلم على الحاسوب} \\
    \hline
  \end{tabularx}\par\vspace{-0.4pt}%
}
\newcommand{\taskrow}[3]{%
  \noindent
  \begin{tabularx}{\textwidth}{@{}>{\centering\arraybackslash}p{2.9cm}%
      >{\raggedleft\arraybackslash}X>{\raggedleft\arraybackslash}X@{}}
    \begin{minipage}[t]{2.7cm}\centering\textbf{\color{primary}#1}\end{minipage} &
    \begin{minipage}[t]{\linewidth}\raggedleft #2\end{minipage} &
    \begin{minipage}[t]{\linewidth}\raggedleft #3\end{minipage} \\
    \hline
  \end{tabularx}\par\vspace{-0.4pt}%
}
