% ── خاص بالوثيقة: عنوان الوثيقة ─────────────────────────────────────────────
\newcommand{\sheettitle}[4]{%
  \begin{center}
    {\Large\bfseries\color{primary} وثيقة التلميذ — الأعمال التطبيقية رقم: #1}
  \end{center}
  \vspace{1pt}
  {\color{secondary}\hrule height 1.2pt}
  \vspace{4pt}
  \begin{center}
    {\large\bfseries\color{primary} #2}
  \end{center}
  \begin{center}
    {\normalsize\color{darktext} المكان: #3 \quad$\cdot$\quad المدة الزمنية: #4}
  \end{center}
  \vspace{2pt}
  {\color{secondary}\hrule height 1.2pt}
  \vspace{6pt}
}

% ── خاص بالوثيقة: بيانات التلميذ ────────────────────────────────────────────
\newcommand{\studentid}[1]{%
  \noindent
  \begin{tabularx}{\textwidth}{@{}>{\bfseries\color{primary}}p{2.6cm}X@{}%
      >{\bfseries\color{primary}}p{2.9cm}p{2.4cm}@{}}
    \hline
    اللقب والاسم: & \blank[5.0cm] &
    القسم / الفوج: & \blank[2.2cm] \\
    \hline
    رقم الجهاز: & \blank[5.0cm] &
    التاريخ: & \blank[2.2cm] \\
    \hline
  \end{tabularx}\par\vspace{8pt}%
}

% ── خاص بالوثيقة: صندوق مهمة ────────────────────────────────────────────────
\newtcolorbox{taskbox}[1][]{
  enhanced, colback=white, colframe=secondary,
  coltitle=white, fonttitle=\bfseries, title=#1,
  boxrule=1pt, arc=4pt, left=10pt, right=10pt, top=8pt, bottom=8pt,
  breakable,
}

% ── خاص بالوثيقة: جدول التدوين المنقّط (tabular عادي ليقبل أسطرًا متعدّدة) ──
\newcommand{\specheader}{%
  \noindent
  \begin{tabular}{@{}>{\raggedleft\arraybackslash}p{6.0cm}@{\hspace{10pt}}%
      >{\raggedleft\arraybackslash}p{9.0cm}@{}}
    \toprule
    \textbf{\color{primary}العنصر المستهدف} &
    \textbf{\color{primary}المواصفات المدوّنة على الجهاز} \\
    \midrule
}
\newcommand{\specrow}[2]{%
  #1 & \specdots{#2} \\
}
\newcommand{\specend}{%
  \bottomrule
  \end{tabular}\par\vspace{8pt}%
}
\newcommand{\specdots}[1]{%
  \makebox[\linewidth]{\dotfill}%
  \ifnum#1>1 \\\makebox[\linewidth]{\dotfill}\fi
  \ifnum#1>2 \\\makebox[\linewidth]{\dotfill}\fi
  \ifnum#1>3 \\\makebox[\linewidth]{\dotfill}\fi
}

% ── خاص بالوثيقة: جدول التقييم (booktabs) ───────────────────────────────────
\newcommand{\gradetable}{%
  \noindent
  \begin{tabularx}{\textwidth}{@{}X>{\centering\arraybackslash}p{3.0cm}@{}}
    \toprule
    \textbf{\color{primary}المعيار} &
    \textbf{\color{primary}الملاحظة والدرجة} \\
    \midrule
    1. التشغيل والاستعمال الآمن للجهاز والالتزام بالنظام. & \phantom{0}/ 04 \\
    2. استخراج مواصفات الجهاز بدقة (المهمة 1). & \phantom{0}/ 05 \\
    3. إدخال البيانات ومعالجتها واستخراج المعلومة الصحيحة (المهمة 2). & \phantom{0}/ 06 \\
    4. إنشاء المجلد وحفظ الملف في المسار الصحيح (المهمة 3). & \phantom{0}/ 05 \\
    \midrule
    \textbf{\color{primary}المجموع النهائي} & \textbf{\phantom{0}/ 20} \\
    \bottomrule
  \end{tabularx}\par\vspace{4pt}%
}
