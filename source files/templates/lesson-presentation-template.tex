% شريط عنوان الشريحة
\setbeamertemplate{frametitle}{%
  \nointerlineskip%
  \hbox to\paperwidth{%
    \hspace*{-1.05cm}%
    \begin{beamercolorbox}[wd=\paperwidth, ht=3.2ex, dp=1.4ex,
        rightskip=1em plus 1fil, leftskip=1em, sep=0pt]{frametitle}%
      \strut\insertframetitle\strut
    \end{beamercolorbox}%
    \hss
  }%
}

% شريط سفلي
\setbeamertemplate{footline}{%
  \hbox{%
    \begin{beamercolorbox}[wd=.34\paperwidth, ht=2.6ex, dp=1.2ex, leftskip=1em]{}%
      \usebeamerfont{footline}\color{rulegray}\insertshorttitle
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=.33\paperwidth, ht=2.6ex, dp=1.2ex, center]{}%
      \usebeamerfont{footline}\color{rulegray}الأستاذ لفقير عبدالجليل
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=.33\paperwidth, ht=2.6ex, dp=1.2ex, rightskip=1em]{}%
      \usebeamerfont{footline}\color{rulegray}\insertframenumber\,/\,\inserttotalframenumber
    \end{beamercolorbox}%
  }%
}

% شريحة عنوان الدرس (plain)
% \lessonframe{نص تمهيدي}
\newcommand{\lessonframe}[1]{%
  \begin{frame}[plain]
    \vfill
    \begin{center}
      {\color{secondary}\large #1}\\[8pt]
      {\color{primary}\Huge\bfseries\inserttitle}
    \end{center}
    \vfill
  \end{frame}%
}

% شريحة قسم
% \sectionframe{عنوان القسم}
\newcommand{\sectionframe}[1]{%
  \begin{frame}[plain]
    \vfill
    \begin{center}
      {\color{white}\colorbox{primary}{\Large\bfseries\;#1\;}}
    \end{center}
    \vfill
  \end{frame}%
}
