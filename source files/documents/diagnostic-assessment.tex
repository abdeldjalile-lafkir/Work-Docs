\input{"../shared/core.tex"}

% ── حزمة الأعمدة المتعددة (خاصة بالاستمارة) ────────────────────────────────

% ── Compact spacing ──────────────────────────────────────────────────────────
\setlength{\parindent}{0pt}
\setlength{\parskip}{1pt}

% ── بدون تذييل ولا ترقيم (كل استمارة مكتفية بذاتها) ─────────────────────────

% ── Form container: one half-page form ───────────────────────────────────────

  after skip=0pt,
}

% ── Section bar (compact) ────────────────────────────────────────────────────
\titleformat{\section}
  {\normalsize\bfseries\color{white}}
  {}
  {0em}
  {\colorbox{primary}}
  [\vspace{0.1em}{\color{primary}\titlerule[0.8pt]}]
\titlespacing*{\section}{0pt}{3pt}{1pt}

% ── Compact lists ────────────────────────────────────────────────────────────
\newlist{ynlist}{enumerate}{1}
\setlist[ynlist, 1]{
  label=\arabic*.\hspace{0.4em},
  left=0em .. 1.2em,
  itemsep=1pt, parsep=0pt, topsep=1pt,
}

\newlist{termlist}{itemize}{1}
\setlist[termlist, 1]{
  label={}, left=0em .. 0em,
  itemsep=0.5pt, parsep=0pt, topsep=1pt,
}

% ── Helpers (خاصة بالاستمارة؛ \cb و \answerline يأتيان من core.tex) ─────────
\newcommand{\yesno}{\hfill\textbf{نعم}~\cb\quad\textbf{لا}~\cb}

% ── Cut line between the two forms ───────────────────────────────────────────
\newcommand{\cutline}{%
  \par\vspace{4pt}%
  \noindent\begin{tikzpicture}
    \draw[dashed, rulegray, line width=0.6pt] (0,0) -- (\linewidth,0);
    \node[fill=white, inner sep=3pt] at (0.5\linewidth,0)
      {\footnotesize\color{rulegray}قصّ هنا};
  \end{tikzpicture}%
  \par\vspace{4pt}%
}

% ── The reusable form content ────────────────────────────────────────────────
\newcommand{\formcontent}{%
  \begin{center}
    {\small\color{secondary} ثانوية الشهيد حكومي العيد أدرار} \\[1pt]
    {\large\bfseries\color{primary} تقويم تشخيصي - مادة المعلوماتية }
  \end{center}

  \vspace{1pt}

  \small
  \textbf{الاسم واللقب:} \dotfill\quad
  \textbf{القسم:} \dotfill

  \vspace{1pt}

  \section{الجزء الأول: أجب بنعم أو لا}
  \begin{multicols}{2}
  \begin{ynlist}
    \item هل تملك هاتفاً ذكياً؟\yesno
    \item هل يتوفر في منزلك حاسوب (مكتبي أو محمول)؟\yesno
    \item هل تملك بريداً إلكترونياً؟\yesno
    \item هل درست المعلوماتية في الطور المتوسط؟\yesno
    \item هل تستعمل الحاسوب أو الهاتف في دروسك؟\yesno
    \item هل تجيد استعمال Word و Excel؟\yesno
  \end{ynlist}
  \end{multicols}

  \section{الجزء الثاني: أسئلة قصيرة}
  \begin{enumerate}[label=\arabic*.\hspace{0.4em}, left=0em .. 1.2em,
                    itemsep=1pt, parsep=0pt, topsep=1pt]
    \item ما الفرق بين العتاد (Hardware) والبرمجيات (Software)؟\answerline\answerline
    \item ما الفرق بين النسخ (Copy) والقص (Cut)؟\answerline\answerline
    \item اذكر برنامجين من حزمة Microsoft Office واستعمال كل واحد منهما.\answerline\answerline
    \item أيّ تطبيق للذكاء الاصطناعي تستعمله؟ وأيّها أفضل في رأيك؟\answerline\answerline
  \end{enumerate}

}
