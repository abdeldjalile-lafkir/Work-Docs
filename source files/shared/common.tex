\definecolor{secondary}{HTML}{2E86C1}    % أزرق متوسط — العناوين الفرعية

% ── Page Border: 1cm from all four edges (إطار بخط رفيع) ────────────────────
\AddToShipoutPictureBG{%
  \begin{tikzpicture}[remember picture, overlay]
    \draw[primary, line width=1.1pt]
      ([xshift=-1cm, yshift=-1cm]current page.north east)
      rectangle
      ([xshift=1cm, yshift=1cm]current page.south west);
  \end{tikzpicture}%
}

% ── Header / Footer ──────────────────────────────────────────────────────────
% الترويسة الرسمية تُدخل من header.tex، والتذييل من footer.tex

%   • من documents/      → \def\shareddir{../shared/}

% ── Section formatting ───────────────────────────────────────────────────────
\titleformat{\section}
  {\large\bfseries\color{primary}}
  {\colorbox{primary}{\color{white}\thesection}}
  {0.7em}{}
  [\vspace{-0.15em}{\color{primary}\titlerule[1.2pt]}]

\titleformat{\subsection}
  {\normalsize\bfseries\color{secondary}}{\thesubsection}{0.5em}{}

% عنوان قسم بدون ترقيم — نص بارز فقط (بلا قاعدة)
\newcommand{\plainsec}[1]{%
  \par\vspace{8pt}%
  {\large\bfseries\color{primary} #1}\par\vspace{5pt}%
}

% ── Helpers ──────────────────────────────────────────────────────────────────
% سطر نقاط للإجابة (يجب أن يكون داخل صندوق ليرتسم)
\newcommand{\answerline}{\par\vspace{1pt}\noindent\makebox[\linewidth]{\dotfill}}
% عدّة أسطر إجابة — لا تُستعمل داخل خلية جدول
\newcommand{\answerlines}[1]{%
  \begingroup\count255=0 \loop\ifnum\count255<#1 \answerline\advance\count255 by 1 \repeat\endgroup}
% فراغ للكتابة على السطر
\newcommand{\blank}[1][3cm]{\underline{\hspace{#1}}}
% مربّع اختيار
\newcommand{\cb}{\raisebox{0.15ex}{\fbox{\rule{0pt}{1.1ex}\hspace{1em}}}}

% ── البطاقة الفنية (جداول booktabs: بلا خطوط عمودية) ────────────────────────
% \cardtitleset يهيّئ القواعد والمسافات داخل جدول البطاقة
\newcommand{\cardsetup}{%
  \renewcommand{\arraystretch}{1.25}%
  \setlength{\arrayrulewidth}{0.6pt}%
  \arrayrulecolor{rulegray}%
}
% صفوف البطاقة: بلا قاعدة لكل صف — القواعد تُرسم بـ \midrule بين المجموعات
\newcommand{\cardrow}[2]{%
  \textbf{\color{primary}#1} & #2 \\
}
\newcommand{\cardrowtwo}[4]{%
  \textbf{\color{primary}#1} & #2 \hfill \textbf{\color{primary}#3:} #4 \\
}
% عنوان المستند — كتلة عنوان أنيقة (بلا قاعدة)
\newcommand{\cardtitle}[1]{%
  \begin{center}
    {\LARGE\bfseries\color{primary} #1}
  \end{center}
  \vspace{8pt}
}

% ── Callout boxes — صناديق أنيقة بشريط لوني جانبي ───────────────────────────
\newtcolorbox{infobox}[1][]{
  enhanced, breakable,
  colback=lightbg, colframe=secondary,
  coltitle=white, fonttitle=\bfseries, title=#1,
  boxrule=0.0pt, leftrule=2.5pt, arc=2pt,
  left=11pt, right=11pt, top=8pt, bottom=8pt,
}
\newtcolorbox{notebox}[1][]{
  enhanced, breakable,
  colback=lightbg, colframe=primary,
  coltitle=white, fonttitle=\bfseries, title=#1,
  boxrule=0.0pt, leftrule=3pt, arc=2pt,
  left=11pt, right=11pt, top=8pt, bottom=8pt,
}
